        function playTransitionSound() {
            try {
                const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                const osc = audioCtx.createOscillator();
                const gainNode = audioCtx.createGain();
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(300, audioCtx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(600, audioCtx.currentTime + 0.1);
                gainNode.gain.setValueAtTime(0.1, audioCtx.currentTime);
                gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);
                osc.connect(gainNode);
                gainNode.connect(audioCtx.destination);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.1);
            } catch(e) {}
        }

                const gainNode = audioCtx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(450, audioCtx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(120, audioCtx.currentTime + 0.08);
                gainNode.gain.setValueAtTime(0.15, audioCtx.currentTime);
                gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.08);
                osc.connect(gainNode);
                gainNode.connect(audioCtx.destination);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.08);
            } catch(e) {}
        }

        function updateTimer() {
            const now = new Date();
            const diff = targetDate - now;

            if (diff <= 0) {
                isUnlocked = true;
                return;
            }

            const days = Math.floor(diff / (1000 * 60 * 60 * 24));
            const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
            const mins = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
            const secs = Math.floor((diff % (1000 * 60)) / 1000);
            
            const timerEl = document.getElementById('timer');
            if(timerEl) {
                if(currentLang === 'ar') timerEl.innerHTML = `${days}يوم : ${hours}س : ${mins}د : ${secs}ث`;
                else if(currentLang === 'en') timerEl.innerHTML = `${days}d : ${hours}h : ${mins}m : ${secs}s`;
                else timerEl.innerHTML = `${days}ڕۆژ : ${hours}کاتژمێر : ${mins}دەقە : ${secs}چرکە`;
            }
        }
        setInterval(updateTimer, 1000);
        updateTimer();

        function initFullscreenBalloons() {
            const container = document.getElementById('fullscreen-balloons-container');
            container.innerHTML = '';
            poppedCount = 0;
            container.classList.add('active');

            for (let i = 1; i <= totalBalloons; i++) {
                const b = document.createElement('div');
                b.className = 'real-balloon';
                b.innerHTML = i; 

                const randomX = Math.random() * (window.innerWidth - 60);
                const randomY = Math.random() * (window.innerHeight - 100) + 30;

                b.style.left = randomX + 'px';
                b.style.top = randomY + 'px';

                const hue = 270 + Math.floor(Math.random() * 30);
                b.style.background = `radial-gradient(circle at 30% 30%, hsl(${hue}, 70%, 65%), hsl(${hue}, 80%, 30%))`;
                b.style.animationDuration = (3 + Math.random() * 2) + 's';
                b.style.animationDelay = (Math.random() * 2) + 's';

                b.addEventListener('click', function(e) {
                    e.stopPropagation();
                    if (!b.classList.contains('popped')) {
                        b.classList.add('popped');
                        poppedCount++;
                        playPopSound();

                        confetti({
                            particleCount: 15,
                            spread: 50,
                            origin: { x: e.clientX / window.innerWidth, y: e.clientY / window.innerHeight }
                        });

                        if (poppedCount === totalBalloons) {
                            container.classList.remove('active');
                            const popup = document.getElementById('congrats-popup');
                            document.getElementById('popup-text').innerText = translations[currentLang].popupSmart;
                            popup.classList.add('show');
                            confetti({ particleCount: 120, spread: 90, origin: { y: 0.5 } });

                            setTimeout(() => {
                                popup.classList.remove('show');
                                setTimeout(() => { goToStep(4); }, 400);
                            }, 2000);
                        }
                    }
                });
                container.appendChild(b);
            }
        }

        function togglePlay(src) {
            let audio = document.getElementById('memoryAudio');
            let playBtn = document.getElementById('playBtn');
            let progressBar = document.getElementById('progressBar');
            let currentTimeEl = document.getElementById('currentTime');
            const t = translations[currentLang];
            if (!audio) return;

            if (audio.paused) {
                if (!bgMusic.paused) {
                    bgMusic.pause();
                    wasBgMusicPlaying = true;
                    bgMusicText.innerText = t.musicPausedTemp;
                }

                audio.play();
                playBtn.innerHTML = "❚❚";
                currentAudio = audio;
                
                audio.ontimeupdate = () => {
                    progressBar.style.width = ((audio.currentTime / audio.duration) * 100) + '%';
                    if(currentTimeEl) currentTimeEl.innerText = formatTime(audio.currentTime);
                };

                audio.onended = () => { 
                    playBtn.innerHTML = "▶"; 
                    progressBar.style.width = '0%'; 
                    if(currentTimeEl) currentTimeEl.innerText = "0:00";
                    if (wasBgMusicPlaying) {
                        bgMusic.play();
                        bgMusicText.innerText = t.musicPlay;
                        wasBgMusicPlaying = false;
                    }
                };
            } else { 
                audio.pause(); 
                playBtn.innerHTML = "▶"; 
                if (wasBgMusicPlaying) {
                    bgMusic.play();
                    bgMusicText.innerText = t.musicPlay;
                    wasBgMusicPlaying = false;
                }
            }
        }

        function seekAudio(event) {
            let audio = document.getElementById('memoryAudio');
            let container = document.getElementById('progressContainer');
            if (!audio || !audio.duration) return;
            const rect = container.getBoundingClientRect();
            const clickPosition = (event.clientX - rect.left) / container.clientWidth;
            audio.currentTime = clickPosition * audio.duration;
        }
